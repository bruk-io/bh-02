"""Two more facts: does a jail die with its embedder (SIGKILL), and can one jail dial another's
channel? Run with `child` to be the embedder that gets killed."""

import os
import signal
import subprocess
import sys
import time

import brig

PY_TREE = str(Path(sys.base_prefix).parent)
SYSTEM = [p for p in ("/usr", "/bin", "/lib", "/lib64", "/etc") if os.path.exists(p)]
POLICY = brig.FsPolicy(binds=tuple(brig.Bind(p, p) for p in [*SYSTEM, PY_TREE, sys.prefix]))

LISTEN = r"""
import socket, sys, time, os
s = socket.socket(socket.AF_UNIX); s.bind(sys.argv[1]); s.listen(8)
open(sys.argv[2], "w").write(str(os.getpid())) if len(sys.argv) > 2 else None
while True:
    c, _ = s.accept()
    try: c.sendall(b"A-answers\n")
    except OSError: pass
    c.close()
"""
DIAL = r"""
import socket, sys
try:
    c = socket.socket(socket.AF_UNIX); c.connect(sys.argv[1]); print("DIALED", c.recv(20))
except Exception as e:
    print("REFUSED", type(e).__name__, e)
"""


def jail_listening(name: str) -> tuple[brig.Jail, str]:
    res = brig.reserve_channels((brig.Channel(name=name),))
    ep = res.reservations[0].endpoint
    j = brig.create_bwrap_jail((sys.executable, "-I", "-c", LISTEN, ep), policy=POLICY, channels=res)
    assert isinstance(brig.await_channel_ready(j, res.reservations[0], timeout_s=10), brig.ChannelReady)
    return j, ep


if sys.argv[1:] == ["child"]:
    j, ep = jail_listening("tethered")
    print(ep, flush=True)
    time.sleep(600)
    raise SystemExit

# 1. cross-jail dial: B (a jail with no channel of its own, same policy) dials A's endpoint
a, ep_a = jail_listening("a")
b = brig.create_bwrap_jail(
    (sys.executable, "-I", "-c", DIAL, ep_a),
    policy=POLICY,
)
print("exit of B:", brig.run_jail(b, timeout_s=10))
brig.destroy_jail(b)
brig.destroy_jail(a)

# 2. tether: an embedder process holding a jail is SIGKILLed; is the jail's listener gone?
child = subprocess.Popen([sys.executable, __file__, "child"], stdout=subprocess.PIPE, text=True)
ep = child.stdout.readline().strip()
print("child jail endpoint:", ep)
os.kill(child.pid, signal.SIGKILL)
child.wait()
time.sleep(1.0)
import socket

s = socket.socket(socket.AF_UNIX)
try:
    s.connect(ep)
    print("TETHER FAIL: the jail outlived its SIGKILLed embedder:", s.recv(20))
except OSError as e:
    print("TETHER OK: nothing answers after the embedder died:", type(e).__name__)
out = subprocess.run(["pgrep", "-af", "tethered"], capture_output=True, text=True).stdout
print("processes still naming the channel:", out.strip() or "none")
